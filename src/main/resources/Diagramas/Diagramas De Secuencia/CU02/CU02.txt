participant Direccion as dir

Conserje -> UI : Ingresar criterios de búsqueda o no y presiona ACEPTAR
        UI -> Gestor : buscarHuespedes
        activate Gestor
        alt : No ingresa ningun criterio  
           Gestor -> DAO : buscarHuespedes 
           activate DAO
        else : Ingresa uno o mas criterios
           Gestor -> DAO : buscarHuesped(criterios)
           
            
        end
            DAO -> Gestor : listaHuespedes
               deactivate DAO
            opt Lista vacia
                Gestor -> UI : Erorr lista vacia
                deactivate Gestor
                
                UI -> Conserje : Mostrar mensaje "Error no existe huesped con ese criterio"
                Conserje -> Conserje :  Ejecutar CU09 "Dar Alta de Huésped"
  
        end
        ' --- Búsqueda y selección de huésped ---
            Gestor -> UI : Mostrar listaHuesped
            UI -> Conserje : Mostrar listaHuesped
            deactivate UI 

  Conserje -> UI : Seleccionar huésped y presionar "Siguiente"

  opt No seleccionó ningún huésped
  
  UI -> Conserje : Ejecutar CU09 "Dar Alta de Huésped"

end

  UI -> Gestor : seleccionarHuesped(huesped)
